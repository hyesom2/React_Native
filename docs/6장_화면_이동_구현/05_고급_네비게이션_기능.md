# 5. 고급 네비게이션 기능

### 5-1&#41; useNavigation과 useRoute Hooks
- 컴포넌트가 Screen으로 등록되지 않았어도 navigation과 route에 접근할 수 있음

```jsx
import { useNavigation, useRoute } from '@react-navigation/native';

// Screen이 아닌 일반 컴포넌트에서 사용
function MyButton() {
  const navigation = useNavigation();

  return (
    <Button
      title="홈으로 가기"
      onPress={() => navigation.navigate('Home')}
    />
  );
}

function DetailInfo() {
  const route = useRoute();
  const { itemId } = route.params;

  return <Text>Item ID: {itemId}</Text>;
}
```

### 5-2&#41; 네비게이션 이벤트
- 화면이 포커스를 얻거나 잃을 때 특정 동작을 수행할 수 있음

```jsx
import { useFocusEffect } from '@react-navigation/native';

function ProfileScreen() {
  const [data, setData] = useState(null);

  // 화면이 포커스될 때마다 데이터 새로고침
  useFocusEffect(
    React.useCallback(() => {
      // 화면이 포커스될 때
      console.log('화면 포커스됨');
      fetchData();

      return () => {
        // 화면이 포커스를 잃을 때 (정리 함수)
        console.log('화면 블러됨');
      };
    }, [])
  );

  const fetchData = async () => {
    const result = await fetch('https://api.example.com/profile');
    setData(await result.json());
  };

  return <View>{/* UI */}</View>;
}
```

**※ 다양한 네비게이션 이벤트**
```jsx
function MyScreen({ navigation }) {
  useEffect(() => {
    const unsubscribeFocus = navigation.addListener('focus', () => {
      console.log('화면 포커스됨');
    });

    const unsubscribeBlur = navigation.addListener('blur', () => {
      console.log('화면 블러됨');
    });

    const unsubscribeBeforeRemove = navigation.addListener('beforeRemove', (e) => {
      // 화면이 제거되기 전
      e.preventDefault();  // 제거 방지

      Alert.alert(
        '정말 나가시겠습니까?',
        '저장하지 않은 변경사항이 있습니다.',
        [
          { text: '취소', style: 'cancel', onPress: () => {} },
          {
            text: '나가기',
            style: 'destructive',
            onPress: () => navigation.dispatch(e.data.action),
          },
        ]
      );
    });

    return () => {
      unsubscribeFocus();
      unsubscribeBlur();
      unsubscribeBeforeRemove();
    };
  }, [navigation]);

  return <View>{/* UI */}</View>;
}
```

### 5-3&#41; 조건부 렌더링 (인증 플로우)
- 로그인 상태에 따라 다른 네비게이션 구조를 보여줌

```jsx
import React, { useState } from 'react';

function App() {
  const [isLoggedIn, setIsLoggedIn] = useState(false);
  const [isLoading, setIsLoading] = useState(true);

  // 초기 로딩 (토큰 확인 등)
  useEffect(() => {
    checkAuth();
  }, []);

  const checkAuth = async () => {
    try {
      const token = await AsyncStorage.getItem('token');
      setIsLoggedIn(!!token);
    } finally {
      setIsLoading(false);
    }
  };

  if (isLoading) {
    return <SplashScreen />;
  }

  return (
    <NavigationContainer>
      <Stack.Navigator screenOptions={{ headerShown: false }}>
        {isLoggedIn ? (
          // 로그인된 사용자: 메인 앱
          <>
            <Stack.Screen name="Main" component={MainTabNavigator} />
            <Stack.Screen name="Profile" component={ProfileScreen} />
          </>
        ) : (
          // 비로그인 사용자: 인증 플로우
          <>
            <Stack.Screen name="Welcome" component={WelcomeScreen} />
            <Stack.Screen name="Login" component={LoginScreen} />
            <Stack.Screen name="Register" component={RegisterScreen} />
          </>
        )}
      </Stack.Navigator>
    </NavigationContainer>
  );
}
```

### 5-4&#41; 딥 링크 (Deep Linking)
- 외부에서 앱의 특정 화면으로 바로 접근함

```jsx
// app.json에 설정 추가
{
  "expo": {
    "scheme": "myapp",
    "web": {
      "linking": {
        "prefixes": [
          "myapp://",
          "https://myapp.com"
        ]
      }
    }
  }
}

// NavigationContainer에 linking 설정
const linking = {
  prefixes: ['myapp://', 'https://myapp.com'],
  config: {
    screens: {
      Home: 'home',
      Profile: 'profile/:userId',
      Details: 'details/:itemId',
    },
  },
};

<NavigationContainer linking={linking}>
  {/* 네비게이터 */}
</NavigationContainer>

// 이제 다음 URL들이 작동합니다:
// myapp://home -> Home 화면
// myapp://profile/123 -> Profile 화면 (userId: 123)
// myapp://details/456 -> Details 화면 (itemId: 456)
```

### 5-5&#41; 네비게이션 상태 접근
```jsx
import { useNavigationState } from '@react-navigation/native';

function MyComponent() {
  // 현재 라우트 이름 가져오기
  const currentRouteName = useNavigationState(state => 
    state.routes[state.index].name
  );

  // 스택 깊이 확인
  const stackDepth = useNavigationState(state => state.routes.length);

  return (
    <View>
      <Text>현재 화면: {currentRouteName}</Text>
      <Text>스택 깊이: {stackDepth}</Text>
    </View>
  );
}
```
