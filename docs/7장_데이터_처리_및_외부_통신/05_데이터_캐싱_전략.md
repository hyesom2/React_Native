# 5. 데이터 캐싱 전략

### 5-1&#41; 캐싱이란?
- **캐싱(Caching)** 은 한 번 가져온 데이터를 저장해두고 재사용하는 기법
- **장점**
  - 네트워크 요청 감소 → 빠른 로딩
  - 데이터 사용량 절약
  - 오프라인에서도 콘텐츠 표시 가능
  - 서버 부하 감소
- **단점**
  - 오래된 데이터를 보여줄 수 있음
  - 저장 공간 사용
  - 복잡한 관리 필요

### 5-2&#41; 메모리 캐싱 (간단한 방법)
```jsx
// cache.js
class SimpleCache {
  constructor() {
    this.cache = {};
  }

  // 데이터 저장
  set(key, data, expiresIn = 300000) { // 기본 5분
    this.cache[key] = {
      data,
      expiresAt: Date.now() + expiresIn
    };
  }

  // 데이터 가져오기
  get(key) {
    const item = this.cache[key];

    if (!item) return null;

    // 만료 확인
    if (Date.now() > item.expiresAt) {
      delete this.cache[key];
      return null;
    }

    return item.data;
  }

  // 캐시 삭제
  remove(key) {
    delete this.cache[key];
  }

  // 전체 캐시 삭제
  clear() {
    this.cache = {};
  }
}

export const cache = new SimpleCache();

// 사용
import { cache } from './cache';

const fetchWithCache = async (url) => {
  // 1. 캐시 확인
  const cached = cache.get(url);
  if (cached) {
    console.log('캐시에서 가져옴');
    return cached;
  }

  // 2. 캐시 없으면 API 호출
  console.log('API 호출');
  const response = await fetch(url);
  const data = await response.json();

  // 3. 결과를 캐시에 저장
  cache.set(url, data);

  return data;
};
```

### 5-3&#41; AsyncStorage 캐싱
```jsx
// asyncCache.js
import AsyncStorage from '@react-native-async-storage/async-storage';

const CACHE_PREFIX = '@cache:';

export const asyncCache = {
  // 캐시 저장
  async set(key, data, expiresIn = 3600000) { // 기본 1시간
    try {
      const item = {
        data,
        expiresAt: Date.now() + expiresIn
      };
      await AsyncStorage.setItem(
        `${CACHE_PREFIX}${key}`,
        JSON.stringify(item)
      );
    } catch (error) {
      console.error('캐시 저장 오류:', error);
    }
  },

  // 캐시 가져오기
  async get(key) {
    try {
      const item = await AsyncStorage.getItem(`${CACHE_PREFIX}${key}`);

      if (!item) return null;

      const { data, expiresAt } = JSON.parse(item);

      // 만료 확인
      if (Date.now() > expiresAt) {
        await this.remove(key);
        return null;
      }

      return data;
    } catch (error) {
      console.error('캐시 읽기 오류:', error);
      return null;
    }
  },

  // 캐시 삭제
  async remove(key) {
    try {
      await AsyncStorage.removeItem(`${CACHE_PREFIX}${key}`);
    } catch (error) {
      console.error('캐시 삭제 오류:', error);
    }
  },

  // 모든 캐시 삭제
  async clearAll() {
    try {
      const keys = await AsyncStorage.getAllKeys();
      const cacheKeys = keys.filter(key => key.startsWith(CACHE_PREFIX));
      await AsyncStorage.multiRemove(cacheKeys);
    } catch (error) {
      console.error('캐시 전체 삭제 오류:', error);
    }
  }
};
```
